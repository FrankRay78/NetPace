\e[1;38;5;11mDESCRIPTION:\e[0m
Network speed tester including server discovery, latency measurement, download 
and upload speed testing.

\e[1;38;5;11mUSAGE:\e[0m
    netpace \e[38;5;8m[OPTIONS]\e[0m \e[38;5;14m[COMMAND]\e[0m

\e[1;38;5;11mOPTIONS:\e[0m
                           \e[38;5;10mDEFAULT\e[0m                                              
    \e[38;5;7m-h, --help\e[0m                        Prints help information.                  
    \e[38;5;7m-v, --version\e[0m                     Prints version information.               
    \e[38;5;7m    --loop\e[0m                        Performs the speed test on continuous     
                                      loop.                                     
    \e[38;5;7m    --count\e[0m                       Stop speed testing after this many times. 
    \e[38;5;7m    --delay\e[0m                       Time between multiple speed tests         
                                      (HH:MM:SS).                               
    \e[38;5;7m    --csv\e[0m                         Display output as a single CSV row (always
                                      includes timestamp).                      
    \e[38;5;7m    --no-csv-header\e[0m               Do not write the header row (CSV output   
                                      only).                                    
                                      An appended --file target that already    
                                      holds rows is never sent the header.      
    \e[38;5;7m    --json\e[0m                        Display output in Json format.            
    \e[38;5;7m    --json-pretty\e[0m                 Display output in Json format (pretty     
                                      print).                                   
    \e[38;5;7m    --minimal\e[0m                     Display the result as a single compact    
                                      line.                                     
                                      Ideal for batch scripts and redirected    
                                      output.                                   
    \e[38;5;7m    --no-latency\e[0m                  Do not perform latency test.              
                                      When used without --server, the first     
                                      available server is selected.             
    \e[38;5;7m    --no-download\e[0m                 Do not perform download test.             
    \e[38;5;7m    --no-upload\e[0m                   Do not perform upload test.               
    \e[38;5;7m    --server\e[0m                      The URL of a specific speed test server.  
                                      'netpace servers -l' will return your     
                                      nearest servers.                          
    \e[38;5;7m-t, --timestamp\e[0m                   Include a timestamp in the output.        
    \e[38;5;7m    --profile\e[0m          \e[1mmedium\e[0m     Profile bundle of payload settings. <tiny,
                                      small, medium, large, mega>               
    \e[38;5;7m    --downloadsize\e[0m                Stop the download test after this many    
                                      megabytes (IEC MiB).                      
    \e[38;5;7m    --uploadsize\e[0m                  Stop the upload test after this many      
                                      megabytes (IEC MiB).                      
    \e[38;5;7m    --connections\e[0m                 Maximum transfers the download and upload 
                                      tests each run at once (1-32).            
                                      Overrides the --profile connection count; 
                                      the profile's data caps still apply.      
                                      A maximum, so a profile with fewer        
                                      transfers to run than this will not reach 
                                      it.                                       
    \e[38;5;7m-u, --unit\e[0m             \e[1mbits\e[0m       The speed unit. <bits, bytes>             
                                      Applies to the default and minimal output.
    \e[38;5;7m    --unit-scale\e[0m       \e[1mauto\e[0m       The speed unit scale. <auto, base, kilo,  
                                      mega, giga, tera, peta>                   
                                      Applies to the default and minimal output.
    \e[38;5;7m    --unit-system\e[0m      \e[1mSI\e[0m         The speed unit system. <SI, IEC>          
                                      Applies to the default and minimal output.
                                      SI steps up in powers of 1000 (KB, MB,    
                                      GB), common in networking,                
                                      while IEC uses powers of 1024 (KiB, MiB,  
                                      GiB), standard in computing and storage.  
    \e[38;5;7m    --device-id\e[0m                   Record a label for this machine in CSV and
                                      JSON output.                              
                                      The CSV column is always present; the JSON
                                      field appears only when a label is given. 
    \e[38;5;7m    --custom-field\e[0m                Record a key=value pair in CSV and JSON   
                                      output.                                   
                                      May be given more than once, and is       
                                      recorded in the order given.              
    \e[38;5;7m-f, --file\e[0m                        Write output to file.                     
    \e[38;5;7m    --file-mode\e[0m        \e[1mappend\e[0m     Determines file output behavior. <append, 
                                      overwrite>                                
    \e[38;5;7m-q, --quiet\e[0m                       Suppress all normal console output (file  
                                      output still works).                      
    \e[38;5;7m    --fail-on\e[0m          \e[1mnone\e[0m       Exit with a non-zero code on a failed     
                                      measurement. <none, total, partial>       
                                      With 'none' the exit code is never        
                                      affected; 'total' triggers when a test is 
                                      all-failed;                               
                                      'partial' triggers on any failed request. 
                                      Fail-fast across --count and --loop.      
    \e[38;5;7m    --diagnostics\e[0m                 Write per-request diagnostic detail about 
                                      the run to the error stream.              
                                      The result is unchanged, so 'netpace      
                                      --json --diagnostics > result.json 2>     
                                      diagnostics.log' keeps the two apart.     

\e[1;38;5;11mCOMMANDS:\e[0m
    \e[38;5;14mservers    \e[0mShow the nearest speed test servers.

\e[1;38;5;11mENVIRONMENT:\e[0m
    \e[38;5;7mNETPACE_THEME\e[0m  Colour theme to draw in. <default, plain, green>

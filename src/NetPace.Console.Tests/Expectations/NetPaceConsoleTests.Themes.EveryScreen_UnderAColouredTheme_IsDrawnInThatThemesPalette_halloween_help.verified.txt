\e[1;38;5;166mDESCRIPTION:\e[0m
Network speed tester including server discovery, latency measurement, download 
and upload speed testing.

\e[1;38;5;166mUSAGE:\e[0m
    netpace \e[38;5;102m[OPTIONS]\e[0m \e[38;5;166m[COMMAND]\e[0m

\e[1;38;5;166mOPTIONS:\e[0m
                              \e[38;5;104mDEFAULT\e[0m                                           
    \e[38;5;104m-h, --help\e[0m                            Prints help information.              
    \e[38;5;104m-v, --version\e[0m                         Prints version information.           
    \e[38;5;104m    --loop\e[0m                            Performs the speed test on continuous 
                                          loop.                                 
    \e[38;5;104m    --count\e[0m                           Stop speed testing after this many    
                                          times.                                
    \e[38;5;104m    --delay\e[0m                           Time between multiple speed tests     
                                          (HH:MM:SS).                           
    \e[38;5;104m    --time-limit\e[0m          \e[1;38;5;166m00:10:00\e[0m    Stop a speed test run after this long 
                                          (HH:MM:SS, up to 23:59:59).           
                                          A run that reaches the limit reports  
                                          what it measured up to then.          
                                          Each run of --count or --loop gets the
                                          limit afresh, and --delay is not      
                                          counted.                              
    \e[38;5;104m    --csv\e[0m                             Display output as a single CSV row    
                                          (always includes timestamp).          
    \e[38;5;104m    --no-csv-header\e[0m                   Do not write the header row (CSV      
                                          output only).                         
                                          An appended --file target that already
                                          holds rows is never sent the header.  
    \e[38;5;104m    --json\e[0m                            Display output in Json format.        
    \e[38;5;104m    --json-pretty\e[0m                     Display output in Json format (pretty 
                                          print).                               
    \e[38;5;104m    --minimal\e[0m                         Display the result as a single        
                                          timestamped line per run.             
                                          Ideal for watching repeated runs with 
                                          --loop or --count.                    
    \e[38;5;104m    --simple\e[0m                          Display the result as one named line  
                                          per figure: Ping, Download, Upload.   
                                          Ideal for batch scripts and line-based
                                          parsing.                              
    \e[38;5;104m    --no-latency\e[0m                      Do not perform latency test.          
                                          When used without --server-url, the   
                                          first available server is selected.   
    \e[38;5;104m    --no-download\e[0m                     Do not perform download test.         
    \e[38;5;104m    --no-upload\e[0m                       Do not perform upload test.           
    \e[38;5;104m    --provider\e[0m            \e[1;38;5;166mookla\e[0m       The kind of server to test against.   
                                          <ookla, librespeed>                   
                                          'librespeed' needs --server-url naming
                                          where LibreSpeed is installed.        
    \e[38;5;104m    --server-url\e[0m                      The URL of a specific speed test      
                                          server.                               
                                          'netpace servers -l' will return your 
                                          nearest servers.                      
                                          With --provider librespeed, the       
                                          address LibreSpeed is installed at.   
    \e[38;5;104m    --profile\e[0m             \e[1;38;5;166mmedium\e[0m      Profile bundle of payload settings.   
                                          <tiny, small, medium, large, mega>    
    \e[38;5;104m    --downloadsize\e[0m                    Stop the download test after this many
                                          megabytes (IEC MiB).                  
    \e[38;5;104m    --uploadsize\e[0m                      Stop the upload test after this many  
                                          megabytes (IEC MiB).                  
    \e[38;5;104m    --connections\e[0m                     Maximum transfers the download and    
                                          upload tests each run at once (1-32). 
                                          Overrides the --profile connection    
                                          count; the profile's data caps still  
                                          apply.                                
                                          A maximum, so a profile with fewer    
                                          transfers to run than this will not   
                                          reach it.                             
    \e[38;5;104m-u, --unit\e[0m                \e[1;38;5;166mbits\e[0m        The speed unit. <bits, bytes>         
                                          Applies to the default, minimal and   
                                          simple output.                        
    \e[38;5;104m    --unit-scale\e[0m          \e[1;38;5;166mauto\e[0m        The speed unit scale. <auto, base,    
                                          kilo, mega, giga, tera, peta>         
                                          Applies to the default, minimal and   
                                          simple output.                        
    \e[38;5;104m    --unit-system\e[0m         \e[1;38;5;166mSI\e[0m          The speed unit system. <SI, IEC>      
                                          Applies to the default, minimal and   
                                          simple output.                        
                                          SI steps up in powers of 1000 (KB, MB,
                                          GB), common in networking,            
                                          while IEC uses powers of 1024 (KiB,   
                                          MiB, GiB), standard in computing and  
                                          storage.                              
    \e[38;5;104m    --device-id\e[0m                       Record a label for this machine in CSV
                                          and JSON output.                      
                                          The CSV column is always present; the 
                                          JSON field appears only when a label  
                                          is given.                             
    \e[38;5;104m    --custom-field\e[0m                    Record a key=value pair in CSV and    
                                          JSON output.                          
                                          May be given more than once, and is   
                                          recorded in the order given.          
    \e[38;5;104m-f, --file\e[0m                            Write output to file.                 
    \e[38;5;104m    --file-mode\e[0m           \e[1;38;5;166mappend\e[0m      Determines file output behavior.      
                                          <append, overwrite>                   
    \e[38;5;104m-q, --quiet\e[0m                           Suppress all normal console output    
                                          (file output still works).            
    \e[38;5;104m    --fail-on\e[0m             \e[1;38;5;166mnone\e[0m        Exit with a non-zero code on a failed 
                                          measurement. <none, total, partial>   
                                          With 'none' the exit code is never    
                                          affected; 'total' triggers when a test
                                          is all-failed;                        
                                          'partial' triggers on any failed      
                                          request. Fail-fast across --count and 
                                          --loop.                               
    \e[38;5;104m    --min-download\e[0m                    Exit 3 when the download speed is     
                                          below this, such as 50Mbps.           
                                          Takes any unit NetPace writes a speed 
                                          in, whatever --unit, --unit-system and
                                          --unit-scale say.                     
                                          Checks one run, so not with --loop,   
                                          --count above 1, or --no-download.    
    \e[38;5;104m    --min-upload\e[0m                      Exit 3 when the upload speed is below 
                                          this, such as 50Mbps.                 
                                          Read the same way as --min-download;  
                                          not with --loop, --count above 1, or  
                                          --no-upload.                          
    \e[38;5;104m    --max-latency\e[0m                     Exit 3 when the idle latency is above 
                                          this, such as 40ms.                   
                                          Milliseconds only; not with --loop,   
                                          --count above 1, or --no-latency.     
    \e[38;5;104m    --diagnostics\e[0m                     Write per-request diagnostic detail   
                                          about the run to the error stream.    
                                          The result is unchanged, so 'netpace  
                                          --json --diagnostics > result.json 2> 
                                          diagnostics.log' keeps the two apart. 
    \e[38;5;104m    --diagnostics-file\e[0m                Write the diagnostic detail to this   
                                          file instead of the error stream.     
                                          Turns diagnostics on by itself, and   
                                          replaces the file on every run.       

\e[1;38;5;166mCOMMANDS:\e[0m
    \e[38;5;166mservers    \e[0mShow the nearest speed test servers.

\e[1;38;5;166mENVIRONMENT:\e[0m
    \e[38;5;104mNETPACE_THEME\e[0m  Colour theme to draw in. <default, plain, green, halloween>
